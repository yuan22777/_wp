06/01.js — 高階函式 (Higher-Order Function) JavaScript 
// 高階函式:把函式當參數傳入
function mathTool(num1, num2, action){
    return action(num1, num2);
}

const plus  = mathTool(10, 5, function(a, b){ return a + b; });
const minus = mathTool(10, 5, function(a, b){ return a - b; });

console.log(plus);   // 15
console.log(minus);  // 5
閉包 (Closure) 範例 JavaScript 
// 閉包:函式記住外層變數
function multiplier(factor) {
    return function(number) {
        return number * factor;
    };
}

const double = multiplier(2);
const triple = multiplier(3);

console.log(double(5));  // 10
console.log(triple(5));  // 15

Callback 範例 JavaScript 
// Callback Pattern:函式作為參數
function fetchData(callback) {
    setTimeout(() => {
        const data = { id: 1, name: "測試資料" };
        callback(null, data);   // 錯誤優先回呼
    }, 1000);
}

fetchData((err, result) => {
    if (err) console.error(err);
    else console.log(result);
});

function findMax(arr) {
  let max = arr[0];
  for (let i = 1; i < arr.length; i++) {
    if (arr[i] > max) {
      max = arr[i];
    }
  }
  return max;
}

const numbers = [23, 56, 12, 99, 45];

08 — 傳值 vs 傳址(陣列 vs 重新賦值)JavaScript
